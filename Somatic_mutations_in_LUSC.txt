#!/bin/bash

# Create a log file to track what happens
LOGFILE="file_copy_log.txt"
echo "File Copy Operation $(date)" > "$LOGFILE"

# Counter variables
total_files=0
copied_files=0
skipped_files=0

# Loop through each directory in the current directory
for dir in */; do
  if [ -d "$dir" ]; then
    echo "Processing directory: $dir"
    echo "Processing directory: $dir" >> "$LOGFILE"
    
    # Find all files in this directory
    find "$dir" -maxdepth 1 -type f | while read -r file; do
      filename=$(basename "$file")
      total_files=$((total_files+1))
      
      # Check if file already exists in parent directory
      if [ -f "./$filename" ]; then
        echo "Skipping existing file: $filename"
        echo "SKIPPED: $filename (already exists in parent)" >> "$LOGFILE"
        skipped_files=$((skipped_files+1))
      else
        # Copy the file since it doesn't exist in parent yet
        cp -v "$file" ./
        echo "COPIED: $file -> ./$filename" >> "$LOGFILE"
        copied_files=$((copied_files+1))
      fi
    done
  fi
done

# Print summary
echo "----------------------------------------"
echo "Copy operation summary:"
echo "Total files processed: $total_files"
echo "Files copied: $copied_files"
echo "Files skipped: $skipped_files"
echo "----------------------------------------"
echo "See $LOGFILE for details"

echo "----------------------------------------" >> "$LOGFILE"
echo "Copy operation summary:" >> "$LOGFILE"
echo "Total files processed: $total_files" >> "$LOGFILE"
echo "Files copied: $copied_files" >> "$LOGFILE"
echo "Files skipped: $skipped_files" >> "$LOGFILE"
