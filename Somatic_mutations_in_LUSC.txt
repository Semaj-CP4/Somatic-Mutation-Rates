#!/bin/bash

# Create a log file to track what happens
LOGFILE="file_copy_log.txt"
echo "File Copy Operation $(date)" > "$LOGFILE"

# Counter variables
total_files=0
copied_files=0
skipped_files=0

# Loop through each directory in the current directory
for dir in */; do
  if [ -d "$dir" ]; then
    echo "Processing directory: $dir"
    echo "Processing directory: $dir" >> "$LOGFILE"
    
    # Find all files in this directory
    find "$dir" -maxdepth 1 -type f | while read -r file; do
      filename=$(basename "$file")
      total_files=$((total_files+1))
      
      # Check if file already exists in parent directory
      if [ -f "./$filename" ]; then
        echo "Skipping existing file: $filename"
        echo "SKIPPED: $filename (already exists in parent)" >> "$LOGFILE"
        skipped_files=$((skipped_files+1))
      else
        # Copy the file since it doesn't exist in parent yet
        cp -v "$file" ./
        echo "COPIED: $file -> ./$filename" >> "$LOGFILE"
        copied_files=$((copied_files+1))
      fi
    done
  fi
done

# Print summary
echo "----------------------------------------"
echo "Copy operation summary:"
echo "Total files processed: $total_files"
echo "Files copied: $copied_files"
echo "Files skipped: $skipped_files"
echo "----------------------------------------"
echo "See $LOGFILE for details"

echo "----------------------------------------" >> "$LOGFILE"
echo "Copy operation summary:" >> "$LOGFILE"
echo "Total files processed: $total_files" >> "$LOGFILE"
echo "Files copied: $copied_files" >> "$LOGFILE"
echo "Files skipped: $skipped_files" >> "$LOGFILE"

#!/bin/bash

# Output file for results
output_file="gene_mutation_rates.txt"

# Initialize the output file with headers
echo "File Name,STAT5A Mutations,TP53 Mutations,Total Mutations,STAT5A Rate (%),TP53 Rate (%)" > "$output_file"

# Loop through compressed MAF files
for compressed_file in *.maf.gz; do
  # Skip if no matching files
  [ -f "$compressed_file" ] || continue
  
  # File name without extension for display
  file_name="${compressed_file%.maf.gz}"
  
  echo "Processing $compressed_file..."
  
  # Count mutations for each gene and total mutations (using on-the-fly processing)
  stat5a_count=$(zcat "$compressed_file" | grep -c "STAT5A")
  tp53_count=$(zcat "$compressed_file" | grep -c "TP53")
  total_mutations=$(zcat "$compressed_file" | grep -v "^#" | wc -l)
  
  # Calculate mutation rates as percentages (with error handling for division by zero)
  if [ "$total_mutations" -gt 0 ]; then
    stat5a_rate=$(echo "scale=4; $stat5a_count * 100 / $total_mutations" | bc)
    tp53_rate=$(echo "scale=4; $tp53_count * 100 / $total_mutations" | bc)
  else
    stat5a_rate="N/A"
    tp53_rate="N/A"
  fi
  
  # Append results to output file
  echo "$file_name,$stat5a_count,$tp53_count,$total_mutations,$stat5a_rate,$tp53_rate" >> "$output_file"
done

echo "Analysis complete. Results saved to $output_file"
